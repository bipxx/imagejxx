```powershell
$ErrorActionPreference = 'Stop'
$repoRoot = (Resolve-Path .).Path
$python = Join-Path $repoRoot '.venv/Scripts/python.exe'
$cmakeDir = 'C:/Program Files/Microsoft Visual Studio/2022/Community/Common7/IDE/CommonExtensions/Microsoft/CMake/CMake/bin'
$cmake = Join-Path $cmakeDir 'cmake.exe'
$ctest = Join-Path $cmakeDir 'ctest.exe'
$imguiSdk = Join-Path $repoRoot '.tools/imgui-sdk'
$runId = 'replace-with-a-fresh-packet-run-id' # Never reuse an accepted run ID.
$reports = Join-Path $repoRoot "build/plan-implementation/$runId"
if (Test-Path -LiteralPath $reports) { throw 'Choose a new run ID' }
New-Item -ItemType Directory -Path $reports | Out-Null
$nativeBuildDir = Join-Path $reports 'native-build' # Fresh stateful test outputs.
$desktopPrefix = Join-Path $reports 'desktop'
function Invoke-Check {
    param([string]$Program, [string[]]$Arguments, [string]$Log)
    $gateStatePath = "$Log.state.json"
    if ((Test-Path -LiteralPath $Log) -or (Test-Path -LiteralPath $gateStatePath)) {
        throw "Preserve existing report: $Log"
    }
    $gateState = [ordered]@{
        program = $Program; arguments = $Arguments; log = $Log
        started_utc = [DateTime]::UtcNow.ToString('o')
        completed_utc = $null; exit_code = $null; state = 'running'; error = $null
    }
    $gateState | ConvertTo-Json -Depth 5 | Set-Content -LiteralPath $gateStatePath -Encoding utf8
    try {
        & $Program @Arguments *> $Log
        $gateState.exit_code = $LASTEXITCODE
        $gateState.state = if ($LASTEXITCODE -eq 0) { 'passed' } else { 'failed' }
    } catch {
        $gateState.state = 'error'
        $gateState.error = $_.Exception.Message
        throw
    } finally {
        $gateState.completed_utc = [DateTime]::UtcNow.ToString('o')
        $gateState | ConvertTo-Json -Depth 5 | Set-Content -LiteralPath $gateStatePath -Encoding utf8
    }
    if ($gateState.exit_code -ne 0) { throw "Check failed; inspect $Log" }
}
$env:Path = "$cmakeDir;$env:Path"
$env:CMAKE_GENERATOR = 'Visual Studio 17 2022'
$env:CMAKE_GENERATOR_PLATFORM = 'x64'
$env:CMAKE_BUILD_PARALLEL_LEVEL = '4'
$pybind11CmakeDir = (& $python -m pybind11 --cmakedir).Trim()
if ($LASTEXITCODE -ne 0 -or -not (Test-Path -LiteralPath $pybind11CmakeDir)) {
    throw 'Resolve pybind11 CMake discovery before building the UI wheel'
}
```
