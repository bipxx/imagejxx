cmake_minimum_required(VERSION 3.18)
project(imagejxx_installed_core_tests LANGUAGES CXX)

if(NOT IMAGEJXX_SDK_ROOT OR NOT IMAGEJXX_SDK_PROFILE_CARRIER)
  message(FATAL_ERROR "Set IMAGEJXX_SDK_ROOT and IMAGEJXX_SDK_PROFILE_CARRIER; generate the carrier with scripts/calibration_sdk_profiles.py")
endif()
file(TO_CMAKE_PATH "${IMAGEJXX_SDK_ROOT}" IMAGEJXX_SDK_ROOT)
file(TO_CMAKE_PATH "${IMAGEJXX_SDK_PROFILE_CARRIER}" IMAGEJXX_SDK_PROFILE_CARRIER)
get_filename_component(IMAGEJXX_SDK_ROOT "${IMAGEJXX_SDK_ROOT}" REALPATH)
set(imagejxx_DIR "${IMAGEJXX_SDK_ROOT}/lib/cmake/imagejxx" CACHE PATH
  "Explicit installed SDK under test" FORCE)
if(NOT EXISTS "${IMAGEJXX_SDK_PROFILE_CARRIER}")
  message(FATAL_ERROR "The SDK profile carrier is unavailable")
endif()
find_package(imagejxx 0.2 CONFIG REQUIRED COMPONENTS core
  PATHS "${IMAGEJXX_SDK_ROOT}/lib/cmake/imagejxx" NO_DEFAULT_PATH)
get_target_property(core_is_imported imagejxx::core IMPORTED)
if(NOT core_is_imported)
  message(FATAL_ERROR "The consumer requires an installed imported core target")
endif()
enable_testing()
add_executable(selection_transform_api ../native/test_selection_transform.cpp)
target_link_libraries(selection_transform_api PRIVATE imagejxx::core)
add_test(NAME installed_selection_transform_api COMMAND selection_transform_api)
add_executable(snapshot_capture_api snapshot_capture.cpp)
target_link_libraries(snapshot_capture_api PRIVATE imagejxx::core)
add_test(NAME installed_snapshot_capture_api COMMAND snapshot_capture_api)
add_executable(job_resource_api job_resources.cpp)
target_link_libraries(job_resource_api PRIVATE imagejxx::core)
add_test(NAME installed_job_resource_api COMMAND job_resource_api)
add_executable(point_presentation_api ../native/test_point_presentation.cpp)
target_link_libraries(point_presentation_api PRIVATE imagejxx::core)
add_test(NAME installed_point_presentation_api COMMAND point_presentation_api)
add_executable(core_api ../native/test_core.cpp)
add_executable(target_analysis_api ../native/test_target_analysis.cpp)
target_link_libraries(target_analysis_api PRIVATE imagejxx::core)
add_test(NAME installed_target_analysis_api COMMAND target_analysis_api)
add_executable(calibration_api ../native/test_imagej_projection_execution.cpp)
add_executable(global_calibration_api ../native/test_global_calibration.cpp)
add_executable(calibration_profiles ../native/test_calibration_sdk_profiles.cpp)
foreach(client IN ITEMS core_api calibration_api global_calibration_api calibration_profiles)
  target_link_libraries(${client} PRIVATE imagejxx::core)
  if(MSVC)
    target_compile_options(${client} PRIVATE /utf-8)
  endif()
endforeach()
add_test(NAME installed_core_api COMMAND core_api)
add_test(NAME installed_calibration_api COMMAND calibration_api)
add_test(NAME installed_global_calibration_api COMMAND global_calibration_api)
add_test(NAME installed_calibration_profiles
  COMMAND calibration_profiles "${IMAGEJXX_SDK_PROFILE_CARRIER}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/import-$<CONFIG>.txt"
  CONTENT "archive=$<TARGET_FILE:imagejxx::core>\ninclude=$<TARGET_PROPERTY:imagejxx::core,INTERFACE_INCLUDE_DIRECTORIES>\n")

# Optional additive analysis client; the original four clients remain registered.
if(IMAGEJXX_SDK_ANALYSIS_CARRIER)
  add_executable(calibrated_analysis ../native/test_calibrated_analysis_sdk.cpp)
  target_link_libraries(calibrated_analysis PRIVATE imagejxx::core)
  add_test(NAME installed_calibrated_analysis
    COMMAND calibrated_analysis "${IMAGEJXX_SDK_ANALYSIS_CARRIER}")
endif()
